cmake_minimum_required(VERSION 3.21)
project(LoopEyes VERSION 0.1.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Qml DBus)
add_library(loopeyesbackend MODULE backend/plugin.cpp)
target_link_libraries(loopeyesbackend PRIVATE Qt6::Core Qt6::Qml Qt6::DBus)
set_target_properties(loopeyesbackend PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/backend")
